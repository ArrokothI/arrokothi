23: /**
24:  * Whether a source line is documentation rather than code.
25:  *
26:  * Deliberately narrow: only a line that is entirely a comment counts. A line that mixes code with a
27:  * trailing comment is treated as code, so a forbidden word in either half is reported. Where two
28:  * readings are defensible this takes the one that only ever adds reports - a too-strict rule costs a
29:  * rewritten comment, while a too-loose one lets the discriminator back in behind a `//`.
30:  */
31: const isCommentLine = (line: string): boolean => {
32:   const trimmed = line.trim();
33:   return trimmed === "" || trimmed.startsWith("*") || trimmed.startsWith("//") || trimmed.startsWith("/*");
34: };

45:   const found: { path: string; line: string; number: number }[] = [];
