175:   describe("K11-R1-ID-01 hidden and missing lookups do equal work", () => {
176:     const countingCaller = (namespace: string, scopes: string[]): { caller: { namespace: string; scopes: readonly string[] }; reads: () => number } => {
177:       let reads = 0;
178:       const proxied = new Proxy(scopes.slice(), {
179:         get(target, property, receiver) {
180:           if (property === "length" || (typeof property === "string" && /^\d+$/.test(property))) reads += 1;
181:           return Reflect.get(target, property, receiver);
182:         },
183:       });
184:       return { caller: { namespace, scopes: proxied as unknown as readonly string[] }, reads: () => reads };
185:     };
