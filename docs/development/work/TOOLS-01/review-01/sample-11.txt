1: /**
2:  * The member names the poisoned-prototype sweep installs (correction DEC-8, contract revision 9).
3:  *
4:  * The sweep asks one runtime question: while a Kernel boundary call runs, does the zone ever
5:  * resolve a member through `Object.prototype` or `Function.prototype`? It can only ask it for the
6:  * names it poisons, so this module states how that name set is derived and why it reaches every
7:  * member the zone can name:
8:  *
9:  * - **Spelled in the zone.** Every name the 13 zone sources spell in a member position: the name of
10:  *   a property access, the literal key of an element access, every member declared by an interface,
11:  *   type literal, class or object literal, and every identifier-like string literal (the zone passes
12:  *   member names to `observeOwn`, `observeField` and `hostMember` as literals, and computed reads
13:  *   such as `BOUNDARY_PREFIX[boundary]` take their key from a literal union). Declared members are
14:  *   included even when no code reads them, so an optional member no code spells yet is still
15:  *   covered. A name only the zone's comments mention is not.
16:  * - **Read by the engine on the zone's behalf.** Operations the zone performs can read members it
17:  *   never spells: conversion to a primitive (`Symbol.toPrimitive`, `toString`, `valueOf`), the
18:  *   iteration protocol, promise assimilation (`then`), `JSON.stringify` (`toJSON`), `instanceof`,
19:  *   species lookups, and the function protocol (`call`, `apply`, `bind`). `ENGINE_PROTOCOL_NAMES` lists
20:  *   them with their reason.
21:  * - **Small indices.** `"0"`–`"15"`: an ordinary read of a position a list does not own walks on to
22:  *   `Object.prototype`. Larger indices are left to the inherited-index tests (K11-R6-STATE-02).
23:  *
24:  * What the set cannot contain is a name the zone builds at run time from something other than a
25:  * literal. The zone builds exactly one kind: list positions (`${index}`), bounded above.
26:  */

72: const IDENTIFIER_LIKE = /^[A-Za-z_$][\w$]*$/;

74: /** Every name one zone source spells in a member position (see the module comment). */
75: function spelledNames(fileName: string, text: string, into: Set<string>): void {
76:   const source = ts.createSourceFile(fileName, text, ts.ScriptTarget.Latest, true);
77:   const nameOf = (name: ts.Node | undefined): string | undefined =>
78:     name !== undefined && (ts.isIdentifier(name) || ts.isStringLiteral(name) || ts.isNumericLiteral(name)) ? name.text : undefined;
79:   const visit = (node: ts.Node): void => {
80:     let name: string | undefined;
81:     if (ts.isPropertyAccessExpression(node)) name = nameOf(node.name);
82:     else if (ts.isElementAccessExpression(node)) name = nameOf(node.argumentExpression);
83:     else if (
84:       ts.isPropertyAssignment(node) ||
85:       ts.isPropertySignature(node) ||
86:       ts.isMethodSignature(node) ||
87:       ts.isPropertyDeclaration(node) ||
88:       ts.isMethodDeclaration(node) ||
89:       ts.isGetAccessorDeclaration(node) ||
90:       ts.isSetAccessorDeclaration(node) ||
91:       ts.isShorthandPropertyAssignment(node) ||
92:       ts.isBindingElement(node)
93:     ) {
94:       name = nameOf(ts.isBindingElement(node) ? (node.propertyName ?? node.name) : node.name);
95:     } else if (ts.isStringLiteral(node) && IDENTIFIER_LIKE.test(node.text)) {
96:       const parent = node.parent;
97:       if (!ts.isImportDeclaration(parent) && !ts.isExportDeclaration(parent)) name = node.text;
98:     }
99:     if (name !== undefined) into.add(name);
100:     ts.forEachChild(node, visit);
101:   };
102:   visit(source);
103: }

112: /** The complete poisoned name set for the zone at `sourceRoot`, spelled names first. */
113: export function zoneMemberNames(sourceRoot: string): (string | symbol)[] {
114:   const spelled = new Set<string>();
115:   for (const file of zoneSources(sourceRoot)) spelledNames(file, readFileSync(file, "utf8"), spelled);
116:   const names: (string | symbol)[] = [...spelled].sort();
117:   for (const [name] of ENGINE_PROTOCOL_NAMES) if (!names.includes(name)) names.push(name);
118:   for (const name of INDEX_NAMES) if (!names.includes(name)) names.push(name);
119:   return names.filter((name) => !NEVER_POISONED.has(name));
120: }
