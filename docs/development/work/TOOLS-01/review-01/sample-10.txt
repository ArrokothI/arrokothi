1: /**
2:  * Poisoned built-in prototypes for the DEC-8 runtime sweep (contract revision 9, amendment 03).
3:  *
4:  * While a Kernel boundary call runs, every name of the sweep's name set is installed on
5:  * `Object.prototype` and `Function.prototype` as an accessor. The zone reads no member its object
6:  * may not own (correction DEC-8), so a correct zone never resolves any of these accessors: every
7:  * member it reads is found on the object itself, or on a host or class prototype below the two
8:  * built-in roots. An accessor that does run was reached by walking past the object, and the sweep
9:  * attributes it:
10:  *
11:  * - **Zone.** The first non-built-in stack frame above the accessor is a zone source file (or the
12:  *   `canonicalize` dependency the zone calls): a Kernel read of a member its object does not own.
13:  *   The mode decides what happens next. `count` records it and answers as the prototype would have;
14:  *   `throw` records it and throws; `reenter` records it, re-enters the Kernel through the window's
15:  *   reentry hook, then answers as the prototype would have.
16:  * - **Caller.** Any other frame (test code, a fake Driver, a caller getter): the accessor records a
17:  *   liveness count and answers exactly as the prototype would have, so caller code sees no change.
18:  * - **Proxy handler.** The receiver is a registered Proxy handler: the engine looking up a trap the
19:  *   caller's handler does not define. That is Proxy semantics of a caller-owned object, induced by an
20:  *   observation the zone is allowed to make, not a Kernel read; it is counted separately.
21:  *
22:  * The accessors are present only between the outermost `enter` and its `exit`, so everything
23:  * outside Kernel calls (the test runner, fixtures, assertions) runs against untouched prototypes. An
24:  * installed name that a caller replaces during the window (a hostile getter installing its own
25:  * pollution) is left as the caller installed it; only accessors still installed are removed.
26:  *
27:  * Everything this module does while a window is open goes through references captured when it
28:  * loaded and null-prototype descriptors, so the sweep does not trip its own poison.
29:  */

489: /**
490:  * A structural digest of a boundary result: own properties only, through their descriptors (so no
491:  * getter runs and no prototype is consulted), with frozen-ness and shared references recorded.
492:  */
493: export function digest(value: unknown): string {
494:   const hash = createHash("sha256");
495:   const seen = new NativeMap<object, number>();
496:   let objects = 0;
497:   const walk = (current: unknown): void => {
498:     if (current === null) return void hash.update("n;");
499:     switch (typeof current) {
500:       case "undefined":
501:         return void hash.update("u;");
502:       case "boolean":
503:         return void hash.update(current ? "t;" : "f;");
504:       case "number":
505:         return void hash.update(`d${ObjectIs(current, -0) ? "-0" : `${current}`};`);
506:       case "bigint":
507:         return void hash.update(`b${current};`);
508:       case "string":
509:         return void hash.update(`s${current.length}:`).update(current);
510:       case "symbol":
511:         return void hash.update("y;");
512:       case "function":
513:         return void hash.update("F;");
514:       default:
515:         break;
516:     }
517:     const object = current as object;
518:     const previous = ReflectApply(MapGet, seen, [object]) as number | undefined;
519:     if (previous !== undefined) return void hash.update(`r${previous};`);
520:     ReflectApply(MapSet, seen, [object, objects]);
521:     objects += 1;
522:     const proto = ObjectGetPrototypeOf(object);
523:     hash.update(`${ArrayIsArray(object) ? "A" : proto === null ? "N" : "O"}${ObjectIsFrozen(object) ? "z" : "m"}{`);
524:     const keys = ReflectOwnKeys(object);
525:     for (let index = 0; index < keys.length; index += 1) {
526:       const key = keys[index] as string | symbol;
527:       const descriptor = ObjectGetOwnPropertyDescriptor(object, key) as PropertyDescriptor;
528:       hash.update(typeof key === "symbol" ? "y=" : `k${key.length}:${key}=`);

531:     }
